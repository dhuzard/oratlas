import {it,vi,expect} from 'vitest';import {writeFileSync}from'node:fs';
import{workerAuthorityHeaders,workerCredential,workerLease}from'./worker-authority.js';import{sha256Text}from'./principals.js';
it('emits reachable artifact/finding fixture states from native calls without exposing capabilities',async()=>{
 vi.useFakeTimers({toFake:['Date'],now:new Date('2026-03-31T23:51:00.000Z')});
 const api=await import('@/lib/scientific-verification');const{prisma}=await import('@/lib/db');
 const bodies={uploaded:{fixture:'synthetic-g1-artifact',role:'uploaded-transition'},completed:{fixture:'synthetic-g1-artifact',role:'completed-reference'},prepared:{fixture:'synthetic-g1-artifact',role:'prepared-replay'},priorOwner:{fixture:'synthetic-g1-artifact',role:'prior-lease-owner'},expired:{fixture:'synthetic-g1-artifact',role:'expired-lease'}};
 const metadata=(key:keyof typeof bodies)=>({artifactKey:'g1-artifact-'+key.toLowerCase(),kind:'synthetic-storage-observation',mediaType:'application/json',sha256:sha256Text(JSON.stringify(bodies[key])),byteLength:Buffer.byteLength(JSON.stringify(bodies[key])),visibility:'private'});
 const credential=workerCredential('verifier','wrong-owner-submit');
 const authRequest=(which:'submit-only'|'wrong-owner-submit',lease?:string)=>new Request('https://g1.oratlas.example/native-artifact-setup',{headers:{...workerAuthorityHeaders({kind:'verifier',credential:which,lease:'missing'}),...(lease?{'x-oratlas-verification-lease':lease}:{})}});
 try{
 await prisma.verifierCredential.create({data:{id:'g1-worker-verifier-wrong-owner-submit',verifierId:'g1-worker-verifier-other',label:'Synthetic other-verifier submit authority',tokenPrefix:credential.tokenPrefix,tokenHash:credential.tokenHash,scopesJson:'["verification:submit"]',expiresAt:new Date('2026-04-02T00:00:00.000Z'),issuedById:'g1-user-admin',createdAt:new Date('2026-03-30T00:00:00.000Z')}});
 const created=await api.createVerificationRun({subject:{type:'publication-version',publicationVersionId:'g1-worker-version'},verificationProtocolId:'g1-verification-protocol-editorial',inputProfile:'full',idempotencyKey:'g1-artifact-native-run'},'g1-user-admin');
 vi.setSystemTime(new Date('2026-03-31T23:52:00.000Z'));let request=authRequest('wrong-owner-submit');let auth=await api.authenticateVerifier(request,'verification:submit');const firstLease=await api.claimVerificationRun(created.id,auth,300);const initialRun=await prisma.verificationRun.findUniqueOrThrow({where:{id:created.id}});
 vi.setSystemTime(new Date('2026-03-31T23:53:00.000Z'));request=authRequest('wrong-owner-submit',firstLease.leaseToken);auth=await api.authenticateVerifier(request,'verification:submit');const priorOwner=await api.prepareVerificationArtifact(created.id,metadata('priorOwner'),auth,request);
 vi.setSystemTime(new Date('2026-03-31T23:56:00.000Z'));request=authRequest('submit-only',workerLease());auth=await api.authenticateVerifier(request,'verification:submit');const expired=await api.prepareVerificationArtifact('g1-worker-run-expired',metadata('expired'),auth,request);
 vi.setSystemTime(new Date('2026-03-31T23:58:00.000Z'));request=authRequest('submit-only');auth=await api.authenticateVerifier(request,'verification:submit');const secondLease=await api.claimVerificationRun(created.id,auth,300);expect(secondLease.status).toBe('claimed');request=authRequest('submit-only',secondLease.leaseToken);await api.transitionVerificationRun(created.id,{status:'running'},{verifier:auth,request});
 vi.setSystemTime(new Date('2026-03-31T23:58:10.000Z'));request=authRequest('submit-only',secondLease.leaseToken);auth=await api.authenticateVerifier(request,'verification:submit');const uploaded=await api.prepareVerificationArtifact(created.id,metadata('uploaded'),auth,request);
 vi.setSystemTime(new Date('2026-03-31T23:58:20.000Z'));auth=await api.authenticateVerifier(request,'verification:submit');await api.uploadVerificationArtifact(uploaded.artifactId,Buffer.from(JSON.stringify(bodies.uploaded)),'application/json',auth,request);
 vi.setSystemTime(new Date('2026-03-31T23:58:30.000Z'));auth=await api.authenticateVerifier(request,'verification:submit');const completed=await api.prepareVerificationArtifact(created.id,metadata('completed'),auth,request);
 vi.setSystemTime(new Date('2026-03-31T23:58:40.000Z'));auth=await api.authenticateVerifier(request,'verification:submit');await api.uploadVerificationArtifact(completed.artifactId,Buffer.from(JSON.stringify(bodies.completed)),'application/json',auth,request);
 vi.setSystemTime(new Date('2026-03-31T23:58:50.000Z'));auth=await api.authenticateVerifier(request,'verification:submit');await api.completeVerificationArtifact(created.id,completed.artifactId,auth,request);
 const finding={findingKey:'g1-artifact-existing-finding',findingType:'synthetic-storage-observation',status:'not-applicable' as const,impact:'informational' as const,statement:'Synthetic storage fixture, not a scientific finding.',rationale:'This fixture exercises an attributed native replay without asserting scientific validity.',evidenceRefs:[{type:'capture' as const,id:'g1-worker-source'}],artifactRefs:[completed.artifactId]};
 vi.setSystemTime(new Date('2026-03-31T23:59:00.000Z'));auth=await api.authenticateVerifier(request,'verification:submit');const submitted=await api.submitVerificationFinding(created.id,finding,auth,request);
 vi.setSystemTime(new Date('2026-03-31T23:59:10.000Z'));auth=await api.authenticateVerifier(request,'verification:submit');const prepared=await api.prepareVerificationArtifact(created.id,metadata('prepared'),auth,request);
 const run=await prisma.verificationRun.findUniqueOrThrow({where:{id:created.id}});expect(run.leaseGeneration).toBe(2);
 const events=await prisma.verificationRunLifecycleEvent.findMany({where:{verificationRunId:run.id},orderBy:{createdAt:'asc'}});
 const artifactIds=[uploaded.artifactId,completed.artifactId,prepared.artifactId,priorOwner.artifactId,expired.artifactId];
 const artifacts=await prisma.verificationArtifact.findMany({where:{id:{in:artifactIds}},orderBy:{id:'asc'}});
 const findingRow=await prisma.verificationFinding.findUniqueOrThrow({where:{id:submitted.id}});
 const joins=await prisma.verificationFindingArtifact.findMany({where:{verificationFindingId:submitted.id}});
 const blobRows=await prisma.$queryRawUnsafe<Array<{rows:unknown[]}>>(`SELECT json_agg(to_jsonb(t) ORDER BY "artifactId") AS rows FROM public."VerificationArtifactBlob" t WHERE "artifactId" IN ('${uploaded.artifactId}','${completed.artifactId}')`);
 const creds=await prisma.verifierCredential.findMany({where:{id:{in:['g1-worker-verifier-submit-only','g1-worker-verifier-wrong-owner-submit']}},orderBy:{id:'asc'}});
 writeFileSync('/tmp/oratlas-g1-artifacts-pg/native-fixture.json',JSON.stringify({run,initialRun,events,artifacts,blobs:blobRows[0]!.rows,finding:findingRow,joins,credentials:creds,requests:{bodies,metadata:Object.fromEntries(Object.keys(bodies).map(k=>[k,metadata(k as keyof typeof bodies)])),finding},ids:{run:run.id,uploaded:uploaded.artifactId,completed:completed.artifactId,prepared:prepared.artifactId,priorOwner:priorOwner.artifactId,expired:expired.artifactId,finding:submitted.id}},null,2)+'\n',{mode:0o600});
 }finally{await prisma.$disconnect();vi.useRealTimers();}
});
