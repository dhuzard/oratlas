import {it,vi,expect} from "vitest";
import {writeFileSync} from "node:fs";
it("produces native open protocol and citation proposals without providers",async()=>{
 vi.useFakeTimers({toFake:["Date"],now:new Date("2026-03-31T23:55:00.000Z")});let outbound=0;vi.stubGlobal("fetch",()=>{outbound++;throw Error("Resolver setup outbound forbidden");});
 const {prisma}=await import("@/lib/db"),p=await import("@/lib/protocol-drift"),m=await import("@/lib/claim-monitoring");
 try {
 const actor={id:"g1-user-admin",role:"ADMIN"};
 const input={reviewVersionId:"g1-version-2",registry:"clinicaltrials-gov"as const,sourceUrl:"https://clinicaltrials.gov/study/NCT09999992",sourceVersion:"synthetic-resolver-version-1",fetchedAt:"2026-03-31T23:55:00.000Z",payload:{protocolSection:{identificationModule:{nctId:"NCT09999992",briefTitle:"Synthetic resolver fixture; no scientific interpretation"},eligibilityModule:{eligibilityCriteria:"Inclusion Criteria:\nSynthetic resolver population"},outcomesModule:{primaryOutcomes:[{measure:"Synthetic resolver outcome"}]}}}};
 const registered=await p.registerProtocolSnapshot(actor,input);
 const protocols=await prisma.protocolDriftProposal.findMany({where:{snapshotId:registered.snapshotId},orderBy:{id:"asc"}});expect(protocols).toHaveLength(2);expect(protocols.every(x=>x.status==="open")).toBe(true);
 const monitoring=[];
 for(const name of ["editor","admin"]) {
 const result=await m.registerCitationStatus(actor,{doi:"10.5555/g1-fixture-citation-1",status:"new-evidence",source:`Synthetic resolver ${name} input`,note:"Synthetic native relation fixture; not an actual citation alert."});
 const proposals=await prisma.claimUpdateProposal.findMany({where:{statusRecordId:result.statusRecordId},orderBy:{id:"asc"}});expect(proposals.length).toBeGreaterThan(0);
 const proposal=proposals.find(x=>x.claimId==="g1-claim-1");expect(proposal).toBeDefined();monitoring.push({result,proposals,selectedId:proposal!.id});
 }
 const targets=[{kind:"protocol",id:protocols[0]!.id,actor:"EDITOR"},{kind:"protocol",id:protocols[1]!.id,actor:"ADMIN"},{kind:"monitoring",id:monitoring[0]!.selectedId,actor:"EDITOR"},{kind:"monitoring",id:monitoring[1]!.selectedId,actor:"ADMIN"}];
 expect(outbound).toBe(0);
 writeFileSync("/tmp/oratlas-g1-proposal-resolvers-pg/native-fixture.json",JSON.stringify({input,registered,protocols,monitoring,targets,outbound},null,2)+"\n",{mode:0o600,flag:"wx"});
 } finally {await prisma.$disconnect();vi.useRealTimers();vi.unstubAllGlobals();}
});
